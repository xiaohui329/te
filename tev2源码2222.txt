local CONFIG_FILE = "TeTraX_Settings.json"
local Saved = {}
pcall(function()
    if isfile and isfile(CONFIG_FILE) then
        local raw = readfile(CONFIG_FILE)
        local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
        if ok and type(data) == "table" then Saved = data end
    end
end)

local function getCfg(key, default)
    local v = Saved[key]
    if v == nil then return default end
    return v
end

local saveQueued = false
local function saveCfg()
    if saveQueued then return end
    saveQueued = true
    task.delay(1, function()
        saveQueued = false
        pcall(function()
            if writefile then
                writefile(CONFIG_FILE, HttpService:JSONEncode(Saved))
            end
        end)
    end)
end

local function setCfg(key, value)
    Saved[key] = value
    saveCfg()
end

local function GetEquippedTool(player)
    if not player or not v3item then return "" end
    local ok, equipped = pcall(function()
        if player == LocalPlayer and v3item.inventory and v3item.inventory.getEquippedItem then
            return v3item.inventory.getEquippedItem()
        end
        if v3item.GetEquipped then
            return v3item.GetEquipped(player)
        end
        return nil
    end)
    if not ok or not equipped then return "" end
    if type(equipped) == "table" then
        local n = equipped.name or equipped.Name
        return type(n) == "string" and n or ""
    elseif type(equipped) == "string" then
        return equipped
    end
    return ""
end

local function newDraw(class, props)
    local d = Drawing.new(class)
    for k, v in pairs(props or {}) do

end

local EspCache = {}

local function espInitPlayer(player)
    if player == LocalPlayer or EspCache[player] then return end
    EspCache[player] = {
        box       = newDraw("Square", { Thickness = 1, Filled = false, Color = EspConfig.BoxColor }),
        healthBg  = newDraw("Square", { Thickness = 1, Filled = true,  Color = EspConfig.HealthBackColor }),
        healthFg  = newDraw("Square", { Thickness = 1, Filled = true,  Color = EspConfig.HealthColor }),
        name      = newDraw("Text",   { Size = 14, Center = true, Outline = true, Color = EspConfig.NameColor }),
        dist      = newDraw("Text",   { Size = 12, Center = true, Outline = true, Color = EspConfig.DistanceColor }),
        tool      = newDraw("Text",   { Size = 12, Center = true, Outline = true, Color = EspConfig.ToolColor }),

end

local function espRemovePlayer(player)
    if not EspCache[player] then return end
    for _, d in pairs(EspCache[player]) do
        pcall(d.Remove, d)
    end
    EspCache[player] = nil
end

local function espHideAll(data)
    for _, d in pairs(data) do d.Visible = false end
end

local function espUpdatePlayer(player)
    local data = EspCache[player]
    if not data then return end

    if not EspConfig.Enabled or not uiReady then
        espHideAll(data)
        return
    end

    local char = player.Character
    if not char then espHideAll(data) return end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or hum.Health <= 0 then
        espHideAll(data)
        return
    end

    local camPos = Camera.CFrame.Position
    local dist = (camPos - hrp.Position).Magnitude
    if dist > EspConfig.MaxDistance then
        espHideAll(data)
        return
    end

    local screenPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
    if not onScreen then
        espHideAll(data)
        return
    end

    local topPos    = Camera:WorldToViewportPoint(hrp.Position + Vector3.new(0, 3, 0))
    local bottomPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
    local height = math.abs(topPos.Y - bottomPos.Y)
    local width  = height / 2

    if EspConfig.ShowBox then
        data.box.Visible  = true
        data.box.Color    = EspConfig.BoxColor
        data.box.Size     = Vector2.new(width, height)
        data.box.Position = Vector2.new(screenPos.X - width/2, screenPos.Y - height/2)
    else
        data.box.Visible = false
    end

    if EspConfig.ShowName then
        data.name.Visible  = true
        data.name.Text     = player.Name
        data.name.Position = Vector2.new(screenPos.X, screenPos.Y - height/2 - 16)
    else
        data.name.Visible = false
    end

    if EspConfig.ShowDistance then
        data.dist.Visible  = true
        data.dist.Text     = string.format("%dm", math.floor(dist))
        data.dist.Position = Vector2.new(screenPos.X, screenPos.Y + height/2 + 4)
    else
        data.dist.Visible = false
    end

    if EspConfig.ShowHealth then
        local hp = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
        data.healthBg.Visible  = true
        data.healthBg.Size     = Vector2.new(3, height)
        data.healthBg.Position = Vector2.new(screenPos.X - width/2 - 6, screenPos.Y - height/2)

        data.healthFg.Visible  = true
        data.healthFg.Size     = Vector2.new(3, height * hp)
        data.healthFg.Position = Vector2.new(screenPos.X - width/2 - 6,
                                             screenPos.Y - height/2 + height * (1 - hp))
    else
        data.healthBg.Visible = false
        data.healthFg.Visible = false
    end

    if EspConfig.ShowTool then
        local toolName = GetEquippedTool(player)
        if toolName ~= "" then
            data.tool.Visible  = true
            data.tool.Text     = toolName
            data.tool.Position = Vector2.new(screenPos.X, screenPos.Y + height/2 + 18)
        else
            data.tool.Visible = false
        end
    else
        data.tool.Visible = false
    end

    if EspConfig.ShowTracer then
        data.tracer.Visible = true
        data.tracer.From    = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
        data.tracer.To      = Vector2.new(screenPos.X, screenPos.Y)
    else
        data.tracer.Visible = false
    end
end

local function CreateTracer(origin, dir, isRainbow)
    if not origin or not dir or dir.Magnitude == 0 then return end
    local a0 = Instance.new("Attachment"); a0.WorldPosition = origin; a0.Parent = workspace.Terrain
    local a1 = Instance.new("Attachment"); a1.WorldPosition = origin + dir.Unit * 1000; a1.Parent = workspace.Terrain
    local beam = Instance.new("Beam")
    beam.Texture = "rbxassetid://446111271"
    beam.Width0 = 1.2; beam.Width1 = 1.2

    if isRainbow then
        beam.Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
            ColorSequenceKeypoint.new(0.25, Color3.fromRGB(0, 255, 0)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 0, 255)),
            ColorSequenceKeypoint.new(0.75, Color3.fromRGB(255, 255, 0)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 255))
        })
    else
        beam.Color = ColorSequence.new(Color3.fromRGB(255,255,255))
    end

    beam.Transparency = NumberSequence.new(0.3)
    beam.Attachment0 = a0; beam.Attachment1 = a1
    beam.FaceCamera = true; beam.LightEmission = 1
    beam.Parent = workspace.Terrain
    Debris:AddItem(a0, 1); Debris:AddItem(a1, 1); Debris:AddItem(beam, 1)
end

local function createTrace(targetPos)
    if not showTrace then return end
    local char = LocalPlayer.Character
    local myPos = char and getRoot(char) and getRoot(char).Position or Vector3.zero
    local startPos = myPos + Vector3.new(math.random(-15,15), math.random(-15,15), math.random(-15,15))
    local dir = targetPos - startPos
    if dir.Magnitude == 0 then return end
    CreateTracer(startPos, dir, RAINBOW_TRACE)
end

local function createRainbowTrace(cf)
    if not cf then return end
    if not showTrace then return end
    CreateTracer(cf.Position, cf.LookVector, true)
end

        time = time + dt
        updateAccumulator = updateAccumulator + dt
        if updateAccumulator < updateRate then return end
        updateAccumulator = 0
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local rootPos = char.HumanoidRootPart.Position
            local humanoid = char:FindFirstChild("Humanoid")
            local heightOffset = humanoid and humanoid.HipHeight / 2 + 0.1 or 0.9
            local pos = rootPos - Vector3.new(0, heightOffset, 0)
            for i, part in ipairs(killAuraCircleParts) do
                local angle = (i / #killAuraCircleParts) * 2 * math.pi
                local x = pos.X + math.cos(angle) * meleeAttackRange
                local z = pos.Z + math.sin(angle) * meleeAttackRange
                part.Position = Vector3.new(x, pos.Y, z)
                part.Orientation = Vector3.new(0, -math.deg(angle), 0)
                local r = math.sin(time + i * 0.1) * 40 + 180
                local g = math.sin(time + i * 0.1 + 2) * 40 + 180
                local b = math.sin(time + i * 0.1 + 4) * 40 + 180
                part.Color = Color3.fromRGB(r, g, b)
            end
        end
    end)
end

function HitLog:InstallHook()
    if self.HookInstalled then return end
    self.HookInstalled = true
    local mt = getrawmetatable(game)
    setreadonly(mt, false)
    local oldNamecall = mt.__namecall
    mt.__namecall = newcclosure(function(selfObj, ...)
        local args = {...}
        local method = getnamecallmethod()
        if method == "FireServer" and self.Monitoring then
            if args[2] == "player" and args[3] then
                local hitPlayerId = args[3].hitPlayerId
                if hitPlayerId then
                    self.recentlyHit[hitPlayerId] = true
                end
            end
        end
        return oldNamecall(selfObj, ...)
    end)
    setreadonly(mt, true)
end

function HitLog:StartMonitoring()
    if self.Monitoring then return end
    self:Init()
    self.Enabled = true
    self.Gui.Enabled = true
    self.Monitoring = true
    self.healthCache = {}
    self.recentlyHit = {}
    self.lastCache = {}
    self:InstallHook()

    self.MonitoringConnection = RunService.RenderStepped:Connect(function()
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer or not player.Character then continue end
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            if not hum then continue end

            local oldHealth = self.healthCache[player.UserId] or hum.Health
            local newHealth = hum.Health
            local damage = oldHealth - newHealth

            if damage > 0 and self.recentlyHit[player.UserId] then
                local dmg = math.floor(damage)
                local hp = math.floor(newHealth)
                local key = player.Name .. "|" .. dmg .. "|" .. hp
                if not self.lastCache[key] then
                    self.lastCache[key] = true
                    local text = self:FormatDamageHit(player.Name, dmg, hp)
                    self:Add(text, key)
                end
                self.recentlyHit[player.UserId] = nil
            end
            self.healthCache[player.UserId] = newHealth
        end
    end)

    self:Add('<font color="rgb(0,255,0)">\ 你妈死了</font>')
end

function HitLog:StopMonitoring()
    if self.MonitoringConnection then
        self.MonitoringConnection:Disconnect()
        self.MonitoringConnection = nil
    end
    self.Monitoring = false
    self.Enabled = false
    if self.Gui then
        self.Gui.Enabled = false
    end
    if self.Container then
        for _, child in ipairs(self.Container:GetChildren()) do
            child:Destroy()
        end
    end
    self.ActiveLogs = {}
    self.healthCache = {}
    self.recentlyHit = {}
    self.lastCache = {}
end

local function startFly()
    if flying or not LocalPlayer.Character then return end
    local char = LocalPlayer.Character
    local hum  = char:WaitForChild("Humanoid")
    local root = char:WaitForChild("HumanoidRootPart")
    flying = true
    hum.PlatformStand = true
    flyBV = Instance.new("BodyVelocity", root)
    flyBV.Name = "FlyBodyVelocity"
    flyBV.MaxForce = Vector3.new(9e9,9e9,9e9)
    flyBV.Velocity = Vector3.new(0,0,0)
    flyBG = Instance.new("BodyGyro", root)
    flyBG.Name = "FlyBodyGyro"
    flyBG.MaxTorque = Vector3.new(9e9,9e9,9e9)
    flyBG.P = 3000
    
    local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled

    flyConnection = RunService.RenderStepped:Connect(function()
        if not flying or not root then return end
        local camera = workspace.CurrentCamera
        local velocity = Vector3.new(0,0,0)
        
        if isMobile then
            if hum.MoveDirection.Magnitude > 0 then
                local localDir = camera.CFrame:VectorToObjectSpace(hum.MoveDirection)
                local forwardAmount = -localDir.Z
                local rightAmount = localDir.X

                velocity = (camera.CFrame.LookVector * forwardAmount + camera.CFrame.RightVector * rightAmount) * flySpeed
            end
        else
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then velocity = velocity + (camera.CFrame.LookVector * flySpeed) end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then velocity = velocity - (camera.CFrame.LookVector * flySpeed) end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then velocity = velocity - (camera.CFrame.RightVector * flySpeed) end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then velocity = velocity + (camera.CFrame.RightVector * flySpeed) end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then velocity = velocity + Vector3.new(0, flySpeed, 0) end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then velocity = velocity - Vector3.new(0, flySpeed, 0) end
        end
        
        flyBV.Velocity = velocity
        if camera then flyBG.CFrame = CFrame.lookAt(root.Position, root.Position + camera.CFrame.LookVector) end
    end)
end

local function buyGunWithMultiTP(gunName, duration)
    duration = duration or 3
    local buyLoc = gunBuyLocations[gunName]
    local root = getRoot(LocalPlayer.Character)
    if not buyLoc or not root then return false end

    refreshItems()
    for _, v in pairs(items) do
        if v.name == gunName then FireServer("equip", v.guid); return true end
    end

    local originalCF = root.CFrame
    local lockConn = RunService.Heartbeat:Connect(function()
        if root and root.Parent then
            root.CFrame = buyLoc
            root.Velocity = Vector3.zero
            root.RotVelocity = Vector3.zero
        end
    end)
    root.CFrame = buyLoc

    local bought = false
    local startTime = tick()
    while tick() - startTime < duration do
        pcall(function() InvokeServer("attemptPurchase", gunName) end)
        task.wait(0.1)
        refreshItems()
        for _, v in pairs(items) do
            if v.name == gunName then bought = true; break end
        end
        if bought then break end
    end

    if lockConn then lockConn:Disconnect() end
    root.CFrame = originalCF

    if bought then
        refreshItems()
        for _, v in pairs(items) do
            if v.name == gunName then FireServer("equip", v.guid); break end
        end
    end
    return bought
end

local localHitArmorTween  = nil
local localHitArmorRecord = {}

local function triggerLocalHitmarker(hitPlayer, guid)
    if not hitPlayer or not guid then return end
    pcall(function()
        local soundName = "hitmarker"
        local armor     = hitPlayer:GetAttribute("armor")
        local crosshair = GUILoader.Get("Crosshair")

        if armor and armor > 0 then
            localHitArmorRecord[hitPlayer.UserId] = guid
            if localHitArmorTween then localHitArmorTween:Cancel() end
            if crosshair then
                crosshair.CrosshairContainer.ArmorIndicator.ImageTransparency      = 0.25
                crosshair.CrosshairContainer.ArmorBreakIndicator.ImageTransparency = 1
                localHitArmorTween = TweenService:Create(
                    crosshair.CrosshairContainer.ArmorIndicator,
                    TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.In),
                    { ImageTransparency = 1 }
                )
                localHitArmorTween:Play()
            end
            soundName = "hitmarkerArmor"

        elseif armor and armor <= 0 and localHitArmorRecord[hitPlayer.UserId] then
            if localHitArmorTween then localHitArmorTween:Cancel() end
            if crosshair then
                crosshair.CrosshairContainer.ArmorIndicator.ImageTransparency      = 1
                crosshair.CrosshairContainer.ArmorBreakIndicator.ImageTransparency = 0.25
                localHitArmorTween = TweenService:Create(
                    crosshair.CrosshairContainer.ArmorBreakIndicator,
                    TweenInfo.new(2.4, Enum.EasingStyle.Sine, Enum.EasingDirection.In),
                    { ImageTransparency = 1 }
                )
                localHitArmorTween:Play()
            end
            localHitArmorRecord[hitPlayer.UserId] = nil
            soundName = "hitmarkerBreakArmor"
        end

        task.delay(0.05, function()
            local s = v3sound.newLocalSound(soundName)
            s.PlaybackSpeed = math.random(80, 120) / 100
            s:Play()
            task.wait(1)
            s:Destroy()
            task.wait(2)
            if localHitArmorRecord[hitPlayer.UserId] == guid then
                localHitArmorRecord[hitPlayer.UserId] = nil
            end
        end)
    end)
end

local lastLocalProjTime = 0

local function attackTarget(target)
    local targetPos = target.Position
    if selectedWeapon == "Gun Kill" or selectedWeapon == "ALL Gun Kill" then
        local player = Players:GetPlayerFromCharacter(target.Parent)
        if not player or player == LocalPlayer then return end
        if isPlayerProtected(player) then return end

        if forceEquipEnabled then
            local guid = getGuid(selectedGun)
            if guid then pcall(function() v3item.inventory.setEquipped(guid) end); task.wait(0.01) end
        end

        local item = v3item.inventory.getEquippedItem()

        if not item or item.type ~= "Gun" then return end

        if item.guid then
            FireServer("reload", item.guid)
        end

        if showTrace then createTrace(targetPos) end
        if showHitSound then playHitSound(targetPos) end

        pcall(function()
            local Event = ReplicatedStorage:FindFirstChild("devv")
                and ReplicatedStorage.devv:FindFirstChild("remoteStorage")
                and ReplicatedStorage.devv.remoteStorage:FindFirstChild("replicateShot")
            if Event then Event:FireServer() end
        end)
        local g = devv.load("GUID")()
        FireServer("replicateProjectiles", item.guid, {{g, target.CFrame}}, item.firemode)

        local isLocalProjectile = (not showTrace) and (not showHitSound)

        if isLocalProjectile then
            local localG
            pcall(function()
                local proj = v3item.projectiles
                if not proj or not proj.replicateProjectiles then return end
                local char = LocalPlayer.Character
                local root = getRoot(char)
                if not root then return end

                local origin = root.Position
                local torsoPart = char:FindFirstChild("UpperTorso")
                    or char:FindFirstChild("Head")
                if torsoPart then origin = torsoPart.Position end

                local dir = targetPos - origin
                if dir.Magnitude < 0.05 then return end

                local now = os.clock()
                if now - lastLocalProjTime < 0.02 then return end
                lastLocalProjTime = now

                local cf = CFrame.lookAt(origin, origin + dir.Unit)
                localG = GUID()
                proj.replicateProjectiles(item.name, {{localG, cf}}, LocalPlayer)
            end)

            if localG then
                task.spawn(function()
                    triggerLocalHitmarker(player, localG)
                end)
            end
        end

        FireServer("projectileHit", g, "player", {
            hitSize = target.Size, hitPart = target, pos = targetPos, hitPlayerId = player.UserId
        })

        if item.ammoManager and not isLocalProjectile then
            item.ammoManager.ammo = item.ammoManager.ammo - 1
        end

    elseif selectedWeapon == "Melee Kill" then
        local player = Players:GetPlayerFromCharacter(target.Parent)
        if player and player ~= LocalPlayer then
            local weaponGuid
            if punchStyle == "knuckles" then
                weaponGuid = getKnucklesGuid()
                if not weaponGuid then return end
            else
                weaponGuid = getGuid("Fists")
                if not weaponGuid then return end
            end
            FireServer("equip", weaponGuid)

local function collectAllValidTargets()
    local targets = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            local head     = player.Character:FindFirstChild("Head")
            local humanoid = player.Character:FindFirstChild("Humanoid")
            if head and humanoid and humanoid.Health > 0 then
                local isTarget = true
                if #targetPlayers > 0 then
                    isTarget = tableFind(targetPlayers, player.Name)
                else
                    if LocalPlayer:IsFriendsWith(player.UserId) then isTarget = false end
                end
                if isTarget
                    and not player.Character:FindFirstChildOfClass("ForceField")
                    and (head.Position - avoidPosition).Magnitude > avoidRadius then
                    table.insert(targets, head)
                end
            end
        end
    end
    return targets
end

local function auraHeartbeat()
    local myChar = LocalPlayer.Character
    if not myChar or not getRoot(myChar) then return end
    local myRoot = getRoot(myChar)
    local targetHead, minDist = nil, math.huge
    local shouldSkipFriend = (#targetPlayers == 0)
    if selectedWeapon == "Melee Kill" and punchStyle == "knuckles" and aurablade then
        if not getKnucklesGuid() then
            task.spawn(function()
                purchaseItem("Brass Knuckles", knuckleBuyLocation)
                if not getKnucklesGuid() then purchaseItem("指虎", knuckleBuyLocation) end
            end)
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Head") then
            local isTarget = false
            if #targetPlayers > 0 then isTarget = tableFind(targetPlayers, player.Name) else isTarget = true end
            if isTarget and not player.Character:FindFirstChildOfClass("ForceField") then
                if shouldSkipFriend and LocalPlayer:IsFriendsWith(player.UserId) then
                else
                    local head = player.Character.Head
                    local dist = (myRoot.Position - head.Position).Magnitude
                    if dist < minDist and (head.Position - avoidPosition).Magnitude > avoidRadius then
                        minDist = dist
                        targetHead = head
                    end
                end
            end
        end
    end
    if aurablade and selectedWeapon == "ALL Gun Kill" then
        local targets = collectAllValidTargets()
        if #targets > 0 then
            allGunKillRotation = allGunKillRotation % #targets + 1
            attackTarget(targets[allGunKillRotation])
        end
        return
    end
    if not targetHead then return end
    if tpplayfb or (selectedWeapon == "Gun Kill" and gunKillAutoTP and aurablade) then
        myRoot.CFrame = targetHead.CFrame * CFrame.new(fbx, fby, fbz)
    end
    if aurablade then
        local now = tick()
        local dist = (myRoot.Position - targetHead.Position).Magnitude
        if selectedWeapon == "Gun Kill" then
            attackTarget(targetHead)
        else
            if now - lastAttack >= 0 and dist <= meleeAttackRange then
                attackTarget(targetHead)
                lastAttack = now
            end
        end
    end
end
local function updateAuraConnection()
    if aurablade or tpplayfb then
        if not auraConnection then auraConnection = RunService.Heartbeat:Connect(function()
            if not uiReady then return end
            auraHeartbeat()
        end) end
    else
        if auraConnection then auraConnection:Disconnect(); auraConnection = nil end
    end
end

function combatTick()
    local char = LocalPlayer.Character
    local humanoid = char and char:FindFirstChild("Humanoid")
    if not humanoid then return end
    if autovest and punchStyle == "knuckles" and not getKnucklesGuid() and not busy then
        busy = true
        purchaseItem("Brass Knuckles", knuckleBuyLocation)
        if not getKnucklesGuid() then purchaseItem("指虎", knuckleBuyLocation) end
        busy = false
    end
    refreshItems()
    local health    = humanoid.Health
    local maxHealth = humanoid.MaxHealth
    local armor     = LocalPlayer:GetAttribute("armor") or 0
    if health >= 30 then
        if returnAfterBuy then
            local needHeal = autohealth and health < maxHealth
            local needVest = autovest and armor <= 0
            if (needHeal or needVest) and not buyingInProgress then
                local now = tick()
                if now - lastHealVestTick >= HEAL_VEST_INTERVAL then
                    lastHealVestTick = now
                    buyingInProgress = true
                    local root = getRoot(char)
                    if root then
                        local originalCF = root.CFrame
                        local buyLocation = needHeal and bandageBuyLocation or vestBuyLocation
                        local itemName    = needHeal and "Bandage" or "Light Vest"
                        task.spawn(function()
                            local r = getRoot(LocalPlayer.Character)
                            if not r then buyingInProgress = false; return end
                            r.CFrame = buyLocation
                            r.Velocity = Vector3.zero
                            r.RotVelocity = Vector3.zero
                            local lockConn = RunService.Heartbeat:Connect(function()
                                local r2 = getRoot(LocalPlayer.Character)
                                if r2 then
                                    r2.CFrame = buyLocation
                                    r2.Velocity = Vector3.zero
                                    r2.RotVelocity = Vector3.zero
                                end
                            end)
                            task.wait(0.3)
                            for i = 1, 10 do
                                refreshItems()
                                local itemGuid = getGuid(itemName)
                                if itemGuid then
                                    FireServer("equip", itemGuid)
                                    FireServer("useConsumable", itemGuid)
                                    FireServer("removeItem", itemGuid)
                                    break
                                else
                                    pcall(function() InvokeServer("attemptPurchase", itemName) end)
                                end
                                task.wait(0.1)
                            end
                            task.wait(0.2)
                            refreshItems()
                            local finalGuid = getGuid(itemName)
                            if finalGuid then
                                FireServer("equip", finalGuid)
                                FireServer("useConsumable", finalGuid)
                                FireServer("removeItem", finalGuid)
                            end
                            task.wait(0.2)
                            if lockConn then lockConn:Disconnect() end
                            local finalRoot = getRoot(LocalPlayer.Character)
                            if finalRoot then
                                pcall(function()
                                    finalRoot.CFrame = originalCF
                                    finalRoot.Velocity = Vector3.zero
                                    finalRoot.RotVelocity = Vector3.zero
                                end)
                            end
                            task.wait(0.1)
                            buyingInProgress = false
                        end)
                    else buyingInProgress = false end
                end
            end
        else
            local root = getRoot(char)
            if root then
                if autohealth and health < maxHealth then
                    root.CFrame = bandageBuyLocation
                    root.Velocity = Vector3.zero
                    root.RotVelocity = Vector3.zero
                    refreshItems()
                    local bandage = getGuid("Bandage")
                    if not bandage then
                        pcall(function() InvokeServer("attemptPurchase", "Bandage") end)
                    else
                        FireServer("equip", bandage); FireServer("useConsumable", bandage); FireServer("removeItem", bandage)
                    end
                elseif autovest and armor <= 0 then
                    root.CFrame = vestBuyLocation
                    root.Velocity = Vector3.zero
                    root.RotVelocity = Vector3.zero
                    refreshItems()
                    local lightGuid = getGuid("Light Vest")
                    if not lightGuid then
                        pcall(function() InvokeServer("attemptPurchase", "Light Vest") end)
                    else
                        FireServer("equip", lightGuid); FireServer("useConsumable", lightGuid); FireServer("removeItem", lightGuid)
                    end
                end
            end
        end
    end
    if autokz and maskAutoBuy and char and not char:FindFirstChild(maskNames[selectedMaskType]) and not maskBuying then
        task.spawn(buyMaskIfNeeded)
    end
    if Auarcuff then
        local myRoot = getRoot(char)
        if myRoot then
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local hum  = player.Character:FindFirstChild("Humanoid")
                    local root = player.Character:FindFirstChild("HumanoidRootPart")
                    if hum and root and hum.Health > 0 and hum.Health < 5 and (myRoot.Position - root.Position).Magnitude <= 30 then
                        if not loadModule("ClientReplicator").Get(player, "cuffed") then
                            local cuffGuid
                            for _, v in pairs(items) do if v.name == "Handcuffs" then cuffGuid = v.guid; break end end
                            if not cuffGuid then InvokeServer("attemptPurchase","Handcuffs")
                            else FireServer("equip", cuffGuid); FireServer("cuffPlayer", player) end
                        end
                    end
                end
            end
        end
    end
    if autoStomp then stompAura() end
    if autoGrab  then grabAura()  end
end

local aaFovLastShow, aaFovLastX, aaFovLastY, aaFovLastDiam
local function aafovRadius(cam, fov)
    local clamped = math.clamp((fov or 0) * 0.5, 0, 89)
    local h = cam and cam.ViewportSize.Y or 0
    if h < 1 then return 0 end
    return h * 0.5 * math.tan(math.rad(clamped)) / math.tan(math.rad(35))
end
local function aaViewportReference(viewportSize)
    if UserInputService.TouchEnabled then
        return viewportSize.X * 0.5, viewportSize.Y * 0.5
    end
    local m = UserInputService:GetMouseLocation()
    return m.X, m.Y
end

local function aaStartFov()
    aaStopFov()
    if not aaFovFrame then return end
    local function render()
        local cam = workspace.CurrentCamera
        local show = AimAssist.Enabled and AimAssist.UseFov and cam ~= nil
        if aaFovLastShow ~= show then
            aaFovLastShow = show
            aaFovFrame.Visible = show
        end
        if not show then return end
        local px, py = aaViewportReference(cam.ViewportSize)
        local diam   = math.max(2, math.floor(aafovRadius(cam, AimAssist.Fov) * 2 + 0.5))
        if aaFovLastX ~= px or aaFovLastY ~= py then
            aaFovLastX, aaFovLastY = px, py
            aaFovFrame.Position = UDim2.fromOffset(px, py)
        end
        if aaFovLastDiam ~= diam then
            aaFovLastDiam = diam
            aaFovFrame.Size = UDim2.fromOffset(diam, diam)
        end
    end
    render()
    aaFovConn = RunService.RenderStepped:Connect(render)
end

local function aaGetAimPart(entry)
    if not entry then return nil end
    local part = AimAssist.Part
    if part == "Head" and entry.head and entry.head.Parent then return entry.head end
    if part == "Body" then
        if entry.upper and entry.upper.Parent then return entry.upper end
        if entry.lower and entry.lower.Parent then return entry.lower end
    end
    if entry.upper and entry.upper.Parent then return entry.upper end
    if entry.head  and entry.head.Parent  then return entry.head  end
    return nil
end

local function aaCheckLos(from, to, targetChar, localChar)
    if aaTgt.LosChar ~= localChar then
        aaTgt.LosChar = localChar
        local gameFolder = workspace:FindFirstChild("Game")
        aaTgt.LosFilter = { localChar, gameFolder and gameFolder:FindFirstChild("Local") or nil }
        aaTgt.LosParams.FilterDescendantsInstances = aaTgt.LosFilter
    end
    local delta = to - from
    local mag   = delta.Magnitude
    if mag <= 0.25 then return true end
    local result = workspace:Raycast(from, delta.Unit * math.min(mag, 1500), aaTgt.LosParams)
    return not result or (result.Instance and result.Instance:IsDescendantOf(targetChar))
end

local AA_BASE_TIME, AA_PING_SCALE, AA_PING_BASE, AA_VEL_SCALE = 0.032, 1000, 0.36, 0.9
local function aaPredict(pos, vel, dist, px, py)
    vel = vel or Vector3.new()
    local coeff = AA_BASE_TIME + (dist or 0) / AA_PING_SCALE * AA_PING_BASE
    return pos + Vector3.new(vel.X * px, vel.Y * py, vel.Z * px) * coeff * AA_VEL_SCALE
end

local function fn125_aimAssist(self, dt)
    pcall(function()
        if not aaReady or not AimAssist.Enabled then return end
        local weapon = aaGetEquipped()
        if not weapon or not weapon.ammoManager then return end
        if (tonumber(weapon.ammoManager.ammo) or 0) <= 0 then return end

        local char = LocalPlayer.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local cam = workspace.CurrentCamera
        if not (hrp and cam) then return end

        local camCF        = cam.CFrame
        local camPos       = camCF.Position
        local viewportSize = cam.ViewportSize
        local refX, refY   = aaViewportReference(viewportSize)

        local range      = math.max(1, AimAssist.Range)
        local radius     = aafovRadius(cam, AimAssist.Fov)
        local useFov     = AimAssist.UseFov
        local normScale  = math.max(1, math.min(viewportSize.X, viewportSize.Y) * 0.5)
        local bestScore  = math.huge
        local bestYaw, bestPitch

        for _, entry in ipairs(aaScanTargets()) do
            local aimPart = aaGetAimPart(entry) or (entry.plr and entry.plr.PrimaryPart)
            if aimPart and aimPart.Parent then
                local partPos = aimPart.Position
                local dist    = (hrp.Position - partPos).Magnitude
                if dist <= range then
                    local predicted = aaPredict(partPos, aimPart.AssemblyLinearVelocity or Vector3.new(),
                        dist, AimAssist.PredictionX, AimAssist.PredictionY)
                    local objSpace  = camCF:PointToObjectSpace(predicted)
                    local mag       = objSpace.Magnitude
                    if mag > 0.35 then
                        local inv = 1 / mag
                        local relX, relY, relZ = objSpace.X * inv, objSpace.Y * inv, -objSpace.Z * inv
                        if relZ > 0.02 then
                            local sp = cam:WorldToViewportPoint(predicted)
                            if sp.Z > 0.05 then
                                local dx = sp.X - refX
                                local dy = sp.Y - refY
                                local sd = math.sqrt(dx * dx + dy * dy)
                                if not useFov or sd <= radius then
                                    local score = (sd / normScale) ^ 2 * 1.55 + (dist / range) ^ 2 * 0.35
                                    if score < bestScore
                                        and aaCheckLos(camPos, partPos, entry.plr.Character, char) then
                                        bestYaw   = math.deg(math.atan2(relX, relZ))
                                        bestPitch = math.deg(math.atan2(relY, relZ))
                                        bestScore = score
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end

        if not bestYaw then return end

        local strength = 1
        local mod = weapon.aimAssistMod
        if type(mod) == "number" then
            strength = mod
        elseif type(mod) == "function" then
            local ok, res = pcall(mod, weapon)
            if ok and type(res) == "number" then strength = res end
        end

        local speedX = AimAssist.SpeedX * strength
        local speedY = AimAssist.SpeedY * strength

        local curX = aaClientReplicator.Get(LocalPlayer, "xAngle")
        local curY = aaClientReplicator.Get(LocalPlayer, "yAngle")
        if type(curX) ~= "number" then curX = 0 end
        if type(curY) ~= "number" then curY = 0 end

        aaClientReplicator.Set(LocalPlayer, "xAngle", (curX - bestYaw * speedX) % 360)
        aaClientReplicator.Set(LocalPlayer, "yAngle", math.clamp(curY + bestPitch * speedY, -80, 80))
    end)
end

    end

local function getClosestPlayer()
    local LocalPlayerCharacter = LocalPlayer.Character
    if not LocalPlayerCharacter or not LocalPlayerCharacter:FindFirstChild("HumanoidRootPart") then return nil end
    local localRoot = LocalPlayerCharacter.HumanoidRootPart
    local AimPoint = Camera.ViewportSize / 2
    local candidates = {}
    for _, Player in ipairs(GetPlayers(Players)) do
        if Player ~= LocalPlayer then
            if #targetPlayers > 0 and not table.find(targetPlayers, Player.Name) then continue end
            local Character = Player.Character
            local Humanoid  = Character and Character:FindFirstChildOfClass("Humanoid")
            if Character and Humanoid and Humanoid.Health > 0 then
                local partForChecks = Character:FindFirstChild(SilentAimSettings.TargetPart)
                    or Character:FindFirstChild("HumanoidRootPart")
                if partForChecks then
                    local physicalDist = (localRoot.Position - partForChecks.Position).Magnitude
                    if physicalDist <= SilentAimSettings.MaxDistance then
                        local ScreenPosition, OnScreen = getPositionOnScreen(partForChecks.Position)
                        if OnScreen then
                            local fovDist = (AimPoint - ScreenPosition).Magnitude
                            if fovDist <= SilentAimSettings.FOVRadius then
                                table.insert(candidates, { character = Character, fov = fovDist, dist = physicalDist })
                            end
                        end
                    end
                end
            end
        end
    end
    if #candidates == 0 then return nil end
    table.sort(candidates, function(a, b) return a.fov < b.fov end)
    return candidates[1].character
end

local lastTargetCharacter = nil

    local currentTargetCharacter = getClosestPlayer()
    if currentTargetCharacter ~= lastTargetCharacter then lockedRandomPart = nil end
    lastTargetCharacter = currentTargetCharacter
    if currentTargetCharacter then
        local humanoid = currentTargetCharacter:FindFirstChildOfClass("Humanoid")
        if humanoid and humanoid.Health > 0 then
            local targetPartName = SilentAimSettings.TargetPart
            if targetPartName == "Random" then
                if not lockedRandomPart or not lockedRandomPart.Parent then
                    lockedRandomPart = currentTargetCharacter[ValidTargetParts[math.random(1, #ValidTargetParts)]]
                end
                currentTargetPart = lockedRandomPart
            else
                currentTargetPart = currentTargetCharacter:FindFirstChild(targetPartName)
                    or currentTargetCharacter:FindFirstChild("HumanoidRootPart")
            end
        else currentTargetPart = nil end
    else currentTargetPart = nil end
end

                end
                Arguments[2] = Ray.new(Arguments[2].Origin, (currentTargetPart.Position - Arguments[2].Origin).Unit * 1000)
                return oldNamecall(unpack(Arguments))
            end
        end
    end
    return oldNamecall(...)
end))

            local userId = tonumber(joinPlr)
            if not userId then
                local ok, result = pcall(function()
                    return Players:GetUserIdFromNameAsync(joinPlr)
                end)
                if ok and result then userId = result end
            end

            if not userId then
                warn("[Join Player] : " .. tostring(joinPlr))
                return
            end

            local fireFn = (Signal and Signal.FireServer) or FireServer
            if type(fireFn) == "function" then
                local ok, err = pcall(function()
                    fireFn("joinServer", userId)
                end)
                if not ok then
                    warn("[Join Player]: " .. tostring(err))
                end
            else
                warn("[Join Player] FireServer")
            end
        end
    })
